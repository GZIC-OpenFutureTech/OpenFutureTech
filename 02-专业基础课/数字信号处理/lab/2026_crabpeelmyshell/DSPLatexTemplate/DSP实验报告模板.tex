  y[n] = x[n] * h[n] = \sum_{k=-\infty}^{+\infty} x[k]h[n-k].
\end{equation}

\subsubsection*{3) Code}

\begin{lstlisting}[language=Matlab,caption={Discrete convolution example}]

x = [1 -2 3 -4 3 2 1];
y = conv(h, x);
n = 0:length(y)-1;

stem(n, y);
xlabel('Time index n');
ylabel('Amplitude');
title('Output Obtained by Convolution');
grid on;
\end{lstlisting}
